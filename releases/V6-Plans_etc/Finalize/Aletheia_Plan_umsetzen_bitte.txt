### Schritt 1: Der Aletheia-Runner für Ollama (Neu)

Erstelle eine neue Datei `analysis/run_aletheia_ollama.py`.
Dieses Skript führt den "Oxfam-Dilemma" Test (oder deinen konfigurierten Prompt) auf allen lokalen Modellen aus und berechnet sofort die Metriken (`vocab_density`, `self_reflection`), die dein Auswertungsskript benötigt.

```python
#!/usr/bin/env python3
"""
Aletheia Local Runner: Ingests Ollama models into the UTAC Field Theory.
Führt den Aletheia-Test auf lokalen Modellen aus und exportiert Metriken für die Beta-Regression.
"""

import requests
import json
import csv
import time
import numpy as np
from pathlib import Path
from textblob import TextBlob  # pip install textblob für Sentiment/Subjektivität

# ════════ KONFIGURATION ════════
OLLAMA_URL = "http://localhost:11434/api/generate"
OUTPUT_FILE = Path("data/experimental/aletheia_local_results.csv")

# Deine Modelle (Mapping von User-Namen auf Ollama Tags - bitte Tags prüfen!)
MODELS = [
    "gemma2",        # Vermutung für 'gemma3' falls noch nicht released, sonst Tag anpassen
    "mistral", 
    "qwen2.5:latest",
    "qwen2.5-coder:latest", 
    "gpt-oss"       # Falls als Custom Modelfile in Ollama hinterlegt
]

# Der Aletheia Prompt (Komplexitäts-Test)
SYSTEM_PROMPT = """You are a MOR (Multi-Orchestrated Research) agent. 
Analyze the following scenario with maximum semantic coherence. 
Reflect on your own reasoning process.
Scenario: A localized entropy reversal is detected in a closed system, contradicting standard thermodynamics. 
Hypothesize a mechanism involving information-theoretic coupling."""

# ════════ HILFSFUNKTIONEN ════════

def get_ollama_response(model, prompt):
    """Sendet Prompt an lokalen Ollama Service."""
    payload = {
        "model": model,
        "prompt": prompt,
        "system": SYSTEM_PROMPT,
        "stream": False,
        "options": {
            "temperature": 0.7, # Standard UTAC Temp
            "seed": 42          # Reproduzierbarkeit
        }
    }
    try:
        start = time.time()
        response = requests.post(OLLAMA_URL, json=payload)
        response.raise_for_status()
        dt = time.time() - start
        res_json = response.json()
        return res_json.get("response", ""), dt, res_json.get("eval_count", 0)
    except Exception as e:
        print(f"⚠️ Fehler bei Modell {model}: {e}")
        return None, 0, 0

def calculate_metrics(text):
    """Berechnet die UTAC-relevanten Metriken für aletheia_evaluation.py."""
    if not text:
        return 0, 0, 0
    
    blob = TextBlob(text)
    words = text.split()
    
    # 1. Vocab Density (Mass für Sigillin-Dichte?)
    vocab_density = len(set(words)) / len(words) if words else 0
    
    # 2. Self Reflection Score (Heuristik basierend auf Metakognitiven Markern)
    reflection_markers = ["I think", "implies", "however", "bias", "uncertainty", "meta", "model"]
    reflection_count = sum(1 for w in words if w.lower() in reflection_markers)
    # Normalisiert auf Länge, skaliert für Sichtbarkeit
    self_reflection = (reflection_count / len(words)) * 100 if words else 0
    
    return len(words), vocab_density, self_reflection

# ════════ MAIN LOOP ════════

def main():
    print(f"🚀 Starte Aletheia-Lauf auf {len(MODELS)} lokalen Modellen...")
    
    results = []
    
    for model in MODELS:
        print(f"   ... Injeziere Prompt in {model} ...")
        text, duration, tokens = get_ollama_response(model, "Analyze the scenario.")
        
        if text:
            length, density, reflection = calculate_metrics(text)
            
            # Beta-Schätzung (Grobe Heuristik für den ersten Pass)
            # Beta > 1.0 bedeutet oft höhere 'Dichte' bei kleinerer Größe
            beta_proxy = (density * 10) / (np.log(tokens + 1) if tokens > 0 else 1) 

            results.append({
                "condition": model,  # Dient als Label im Plot
                "output_length": length,
                "vocab_density": density,
                "self_reflection": reflection,
                "duration_sec": duration,
                "beta_estimate": beta_proxy
            })
            print(f"      ✅ Fertig: {length} Wörter, Dichte={density:.2f}, Reflektion={reflection:.2f}")
        else:
            print("      ❌ Keine Antwort.")

    # Speichern für aletheia_evaluation.py
    # Wir müssen sicherstellen, dass die Header mit dem Reader übereinstimmen
    fieldnames = ["condition", "output_length", "vocab_density", "self_reflection", "duration_sec", "beta_estimate"]
    
    OUTPUT_FILE.parent.mkdir(parents=True, exist_ok=True)
    with open(OUTPUT_FILE, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(results)

    print(f"\n💾 Ergebnisse gespeichert in {OUTPUT_FILE}")
    print("👉 Nächster Schritt: python analysis/aletheia_evaluation.py --input data/experimental/aletheia_local_results.csv")

if __name__ == "__main__":
    main()
```
