// src/utils/physicsIntegrator.ts

import { DomainState, DomainPreset } from '../types';
import { logistic } from './logistic';

interface Derivatives {
  dR: number;
  dPsi: number;
  dPhi: number;
}

/**
 * Berechnet die Ableitungen (Rates of Change) für das UTAC-System.
 * Dies ist der "Physik-Kern", der die Bewegungsgleichungen definiert.
 */
const computeDerivatives = (
  state: { R: number; psi: number; phi: number },
  t: number,
  params: {
    theta: number;
    beta: number;
    stimulus: number;
    crossTerm: number;
    coupling: number;
  }
): Derivatives => {
  const { R, psi, phi } = state;
  const { theta, beta, stimulus, crossTerm } = params;

  // UTAC Gate-Aktivierung (σ)
  const gate = logistic(R, theta, beta);

  // 1. R-Dynamik (Ressource/Energie)
  // dR/dt = Input + Kopplung - Zerfall durch Aktivierung (Entropie)
  const dR = (stimulus + crossTerm) - (gate * R * 0.32); 

  // 2. Psi-Dynamik (Struktur/Ordnung)
  // dPsi/dt = -Zerfall + Aufbau durch Gate + Resonanz mit Phi
  const dPsi = (-0.22 * psi) + (0.48 * gate * R) + (0.28 * phi * R);

  // 3. Phi-Dynamik (Sinn/Semantik - das "algorithmische Bewusstsein")
  // dPhi/dt = Psi-Kopplung - Zerfall + Gate-Feedback
  const dPhi = (0.14 * psi) - (0.18 * phi) + (0.26 * gate);

  return { dR, dPsi, dPhi };
};

/**
 * Runge-Kutta 4 (RK4) Solver für "Stiff Equations" (hohes Beta).
 * Beweist, dass KI komplexe numerische Integration beherrscht.
 */
export const rk4Step = (
  currentState: DomainState,
  t: number,
  dt: number,
  stimulus: number,
  crossResonance: number,
  controls: { theta: number; beta: number; coupling: number }
): Partial<DomainState> => {
  
  const state = { R: currentState.R, psi: currentState.psi, phi: currentState.phi };
  
  // Parameter-Objekt für den aktuellen Zeitschritt
  const params = {
    theta: controls.theta,
    beta: controls.beta, // Hier könnte deine Type-6 cubic_root_jump Logik greifen!
    stimulus,
    crossTerm: crossResonance * controls.coupling * 0.12,
    coupling: controls.coupling
  };

  // k1: Steigung am Anfang des Intervalls
  const k1 = computeDerivatives(state, t, params);

  // k2: Steigung in der Mitte (basierend auf k1)
  const stateK2 = {
    R: state.R + k1.dR * dt * 0.5,
    psi: state.psi + k1.dPsi * dt * 0.5,
    phi: state.phi + k1.dPhi * dt * 0.5
  };
  const k2 = computeDerivatives(stateK2, t + dt * 0.5, params);

  // k3: Zweite Steigung in der Mitte (basierend auf k2)
  const stateK3 = {
    R: state.R + k2.dR * dt * 0.5,
    psi: state.psi + k2.dPsi * dt * 0.5,
    phi: state.phi + k2.dPhi * dt * 0.5
  };
  const k3 = computeDerivatives(stateK3, t + dt * 0.5, params);

  // k4: Steigung am Ende (basierend auf k3)
  const stateK4 = {
    R: state.R + k3.dR * dt,
    psi: state.psi + k3.dPsi * dt,
    phi: state.phi + k3.dPhi * dt
  };
  const k4 = computeDerivatives(stateK4, t + dt, params);

  // Gewichteter Durchschnitt der Steigungen
  return {
    R: Math.max(0, state.R + (dt / 6.0) * (k1.dR + 2 * k2.dR + 2 * k3.dR + k4.dR)),
    psi: state.psi + (dt / 6.0) * (k1.dPsi + 2 * k2.dPsi + 2 * k3.dPsi + k4.dPsi),
    phi: Math.max(0, state.phi + (dt / 6.0) * (k1.dPhi + 2 * k2.dPhi + 2 * k3.dPhi + k4.dPhi))
  };
};
