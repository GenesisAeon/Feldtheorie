import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D
from matplotlib.animation import FuncAnimation

class TesseractTimeSlices:
    def __init__(self, resolution=50, num_slices=100):
        """4D-Hyperkubus mit Zeitscheiben"""
        self.resolution = resolution
        self.num_slices = num_slices
        
        # 4D-Block: [x, y, z, t]
        self.block_4d = np.zeros((resolution, resolution, resolution, num_slices))
        
        # Kosmische Konstanten
        self.alpha_inv = 137.036
        self.phi = 1.618034
        
        self.initialize_4d_field()
    
    def initialize_4d_field(self):
        """Implosives Feld im 4D-Block"""
        # Koordinaten
        x = np.linspace(-1, 1, self.resolution)
        y = np.linspace(-1, 1, self.resolution)
        z = np.linspace(-1, 1, self.resolution)
        t = np.linspace(0, 1, self.num_slices)
        
        X, Y, Z, T = np.meshgrid(x, y, z, t, indexing='ij')
        
        # Radiale Distanz im 3D-Raum (für jede Zeit)
        R = np.sqrt(X**2 + Y**2 + Z**2)
        
        # Implosives Feld: Kollaps zum Zentrum über Zeit
        # ψ(r, t) = exp(-α⁻¹ · r² / (1 + t))
        self.block_4d = np.exp(-self.alpha_inv * R**2 / (1 + 10*T))
        
        # Tetraedrische Modulation (optional)
        # self.block_4d *= self.tetrahedral_symmetry(X, Y, Z)
    
    def extract_timeslice(self, t_index):
        """Extrahiere 3D-Kubus bei festem t"""
        # Das ist der Slice: block_4d[:, :, :, t_index]
        cube_3d = self.block_4d[:, :, :, t_index]
        return cube_3d
    
    def render_normal_cube(self, cube_3d, ax, time_label=""):
        """Rendere 3D-Kubus in NORMALER Orientierung (nicht gedreht)"""
        ax.clear()
        
        # Isosurface bei Schwellenwert
        from skimage import measure
        threshold = 0.5 * np.max(cube_3d)
        
        try:
            verts, faces, _, _ = measure.marching_cubes(cube_3d, level=threshold)
            
            # WICHTIG: Keine Rotation! Kubus steht normal
            # Standard-Orientierung: x-rechts, y-oben, z-raus
            ax.plot_trisurf(verts[:, 0], verts[:, 1], faces, verts[:, 2],
                           cmap='viridis', alpha=0.7, linewidth=0)
        except:
            pass
        
        # Kubus-Rahmen (8 Ecken, 12 Kanten)
        self.draw_cube_wireframe(ax, self.resolution)
        
        ax.set_xlabel('X')
        ax.set_ylabel('Y')
        ax.set_zlabel('Z')
        ax.set_title(f'Zeitscheibe: {time_label}')
        
        # NORMAL stehend: gleiche Skalierung alle Achsen
        ax.set_box_aspect([1,1,1])
    
    def draw_cube_wireframe(self, ax, size):
        """Zeichne Kubus-Kanten (nicht Pyramide!)"""
        # 8 Ecken eines Würfels
        s = size - 1
        corners = np.array([
            [0, 0, 0], [s, 0, 0], [s, s, 0], [0, s, 0],  # Untere Fläche
            [0, 0, s], [s, 0, s], [s, s, s], [0, s, s]   # Obere Fläche
        ])
        
        # 12 Kanten
        edges = [
            [0,1], [1,2], [2,3], [3,0],  # Unten
            [4,5], [5,6], [6,7], [7,4],  # Oben
            [0,4], [1,5], [2,6], [3,7]   # Vertikal
        ]
        
        for edge in edges:
            points = corners[edge]
            ax.plot3D(*points.T, 'k-', linewidth=0.5, alpha=0.3)
    
    def animate_timeslices(self, save_path='timeslice_cubes.mp4'):
        """Animiere durch alle Zeitscheiben"""
        fig = plt.figure(figsize=(10, 10))
        ax = fig.add_subplot(111, projection='3d')
        
        def update(frame):
            t_index = frame
            cube_3d = self.extract_timeslice(t_index)
            time = t_index / self.num_slices
            self.render_normal_cube(cube_3d, ax, f"t = {time:.2f}")
            return ax,
        
        anim = FuncAnimation(fig, update, frames=self.num_slices, 
                           interval=50, blit=False)
        
        plt.tight_layout()
        if save_path:
            anim.save(save_path, writer='ffmpeg', fps=20)
        plt.show()
    
    def create_dual_view(self, t_index):
        """Zeige 4D-Block UND 3D-Slice gleichzeitig"""
        fig = plt.figure(figsize=(16, 7))
        
        # Links: 4D-Block (projiziert auf 3D)
        ax1 = fig.add_subplot(121, projection='3d')
        self.visualize_4d_projection(ax1, t_index)
        
        # Rechts: Extrahierter 3D-Kubus (normal stehend)
        ax2 = fig.add_subplot(122, projection='3d')
        cube_3d = self.extract_timeslice(t_index)
        self.render_normal_cube(cube_3d, ax2, f"Slice {t_index}")
        
        plt.tight_layout()
        plt.savefig(f'/mnt/user-data/outputs/timeslice_{t_index:03d}.png', dpi=150)
        plt.show()
    
    def visualize_4d_projection(self, ax, current_t):
        """Projiziere 4D-Hyperkubus auf 3D (alle Zeitscheiben)"""
        # Zeige mehrere Scheiben übereinander (Schatten)
        for t_idx in range(0, self.num_slices, 10):
            cube_3d = self.extract_timeslice(t_idx)
            
            # Nur Kontur
            alpha = 0.1 if t_idx != current_t else 0.8
            color = 'gray' if t_idx != current_t else 'red'
            
            # Würfel-Umriss
            self.draw_cube_wireframe_at_height(ax, cube_3d, t_idx, alpha, color)
        
        ax.set_title('4D-Hyperkubus (alle Zeitscheiben)')
        ax.set_xlabel('X')
        ax.set_ylabel('Y')
        ax.set_zlabel('Z (+ Zeit)')
    
    def draw_cube_wireframe_at_height(self, ax, cube, t_idx, alpha, color):
        """Zeichne Kubus-Outline bei z=t_idx"""
        s = self.resolution - 1
        z_offset = t_idx * 0.05  # Vertikaler Offset für Zeitstapelung
        
        corners = np.array([
            [0, 0, z_offset], [s, 0, z_offset], 
            [s, s, z_offset], [0, s, z_offset],
            [0, 0, z_offset+s*0.05], [s, 0, z_offset+s*0.05],
            [s, s, z_offset+s*0.05], [0, s, z_offset+s*0.05]
        ])
        
        edges = [[0,1],[1,2],[2,3],[3,0],[4,5],[5,6],[6,7],[7,4],
                 [0,4],[1,5],[2,6],[3,7]]
        
        for edge in edges:
            points = corners[edge]
            ax.plot3D(*points.T, color=color, linewidth=1, alpha=alpha)

class PhotonBetweenSlices:
    def __init__(self, tesseract):
        self.tesseract = tesseract
    
    def propagate_photon(self, start_xyz, start_t):
        """Photon startet in Scheibe t, propagiert zu t+1"""
        path_4d = []
        
        x, y, z = start_xyz
        k_vec = np.array([0.1, 0.1, 0.0])  # Richtung im 3D
        
        for t_idx in range(start_t, self.tesseract.num_slices):
            # Photon in aktueller Zeitscheibe
            cube_3d = self.tesseract.extract_timeslice(t_idx)
            
            # Standard-Propagation
            x += k_vec[0]
            y += k_vec[1]
            z += k_vec[2]
            
            # ABER: Kopplung ans Implosionsfeld
            field_strength = cube_3d[int(x), int(y), int(z)]
            k_vec *= (1 - 0.01 * field_strength)  # Abbremsung
            
            path_4d.append([x, y, z, t_idx])
            
            # Randcheck
            if x < 0 or x >= self.tesseract.resolution: break
        
        return np.array(path_4d)
    
    def visualize_light_path(self):
        """Zeige Lichtbahn durch Zeitscheiben"""
        fig = plt.figure(figsize=(12, 10))
        ax = fig.add_subplot(111, projection='3d')
        
        # Mehrere Photonen
        for start_t in [0, 20, 40]:
            path = self.propagate_photon([25, 25, 25], start_t)
            ax.plot3D(path[:, 0], path[:, 1], path[:, 3],  # x, y, t als "z"
                     linewidth=2, label=f'Photon @ t={start_t}')
