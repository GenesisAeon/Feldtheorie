```python
import pandas as pd
import numpy as np
from abc import ABC, abstractmethod

class UtacDataAdapter(ABC):
    """
    Basis-Klasse für alle Real-Data-Quellen.
    Zwingt alle Daten in das Format: [timestamp, value_R, threshold_Theta]
    """

    def __init__(self, source_name):
        self.source_name = source_name

    @abstractmethod
    def fetch_data(self):
        """Muss von der Unterklasse implementiert werden (API call oder File read)"""
        pass

    def normalize(self, raw_data, R_col, Theta_col=None, Theta_static=None):
        """
        Normalisiert R auf die Schwelle Theta (R/Theta).
        Gibt ein DataFrame zurück, das UTAC direkt fressen kann.
        """
        df = raw_data.copy()
        
        # Wenn Theta dynamisch im Datensatz ist
        if Theta_col:
            df['R_normalized'] = df[R_col] / df[Theta_col]
            df['Theta_current'] = df[Theta_col]
        # Wenn Theta statisch ist (z.B. 1.5°C Klima-Ziel)
        elif Theta_static:
            df['R_normalized'] = df[R_col] / Theta_static
            df['Theta_current'] = Theta_static
        else:
            raise ValueError("Entweder Theta_col oder Theta_static muss definiert sein.")
            
        return df[['timestamp', 'R_normalized', 'Theta_current']]

# --- Beispiel Implementierung für USGS Erdbeben CSV ---
class SeismicAdapter(UtacDataAdapter):
    def fetch_data(self, file_path):
        # Liest echte USGS Daten
        df = pd.read_csv(file_path)
        # Konvertiere Zeitstempel
        df['timestamp'] = pd.to_datetime(df['time'])
        return df
```

```python
import requests
import pandas as pd
import numpy as np
from datetime import datetime, timedelta

class UsgsSeismicAdapter:
    """
    Zieht Live-Erdbebendaten vom USGS und bereitet sie für UTAC vor.
    """
    BASE_URL = "https://earthquake.usgs.gov/fdsnws/event/1/query"

    def fetch_live_data(self, days_back=30, min_magnitude=4.5):
        """Holt Daten der letzten X Tage via API."""
        start_time = (datetime.utcnow() - timedelta(days=days_back)).isoformat()
        params = {
            "format": "csv",
            "starttime": start_time,
            "minmagnitude": min_magnitude,
            "orderby": "time"
        }
        
        try:
            response = requests.get(self.BASE_URL, params=params)
            response.raise_for_status()
            # Temporäre Datei speichern oder direkt in Pandas laden
            from io import StringIO
            df = pd.read_csv(StringIO(response.text))
            return self._normalize(df)
        except Exception as e:
            print(f"Fehler beim Datenabruf: {e}")
            return None

    def _normalize(self, df):
        """
        Wandelt Rohdaten in UTAC-Format: R (Magnitude), Theta (lokal?), Zeit.
        Hier vereinfacht: Wir nehmen Theta = 8.0 (katastrophal) als globale Referenz.
        """
        if df is None or df.empty:
            return None
            
        # Zeitstempel konvertieren
        df['timestamp'] = pd.to_datetime(df['time'])
        
        # R = Magnitude (logarithmisch, Energie ~ 10^(1.5*mag))
        # UTAC braucht eine "Fortschrittsvariable". 
        # Option A: R = Magnitude / Theta
        # Option B: R = Energie / Energie(Theta)
        
        # Wir nehmen hier R = Magnitude (einfach)
        theta_global = 8.0
        df['R_normalized'] = df['mag'] / theta_global
        df['Theta_static'] = theta_global
        
        return df[['timestamp', 'latitude', 'longitude', 'mag', 'R_normalized', 'Theta_static']]
```
